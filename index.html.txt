  <script>
    // Initial Seed Data jika LocalStorage kosong
    const initialData = [
      { id: '1', name: 'Elecsys TSH (200 Tests)', category: 'Reagent Kit', lot: '631245', quantity: 6, minThreshold: 3, expiry: '2027-05-18', storage: '2-8°C', location: 'Chiller A - Rak 1' },
      { id: '2', name: 'Elecsys Free T4 (200 Tests)', category: 'Reagent Kit', lot: '639812', quantity: 2, minThreshold: 3, expiry: '2026-11-20', storage: '2-8°C', location: 'Chiller A - Rak 1' },
      { id: '3', name: 'Troponin I STAT (100 Tests)', category: 'Reagent Kit', lot: '710034', quantity: 4, minThreshold: 2, expiry: '2026-12-05', storage: '2-8°C', location: 'Chiller B - Rak 2' },
      { id: '4', name: 'PreciControl Cardiac (Level 1,2,3)', category: 'Quality Control', lot: 'QC-9921', quantity: 1, minThreshold: 2, expiry: '2027-01-30', storage: '-20°C', location: 'Freezer 1' },
      { id: '5', name: 'Universal ProCell / CleanCell Buffer', category: 'Buffer / Consumable', lot: 'B-4401', quantity: 12, minThreshold: 5, expiry: '2027-09-10', storage: '15-25°C', location: 'Stor Bahan Kimia' }
    ];

    let inventory = JSON.parse(localStorage.getItem('immuno_inventory')) || initialData;

    function saveInventory() {
      localStorage.setItem('immuno_inventory', JSON.stringify(inventory));
      renderTable();
    }

    function renderTable() {
      const tbody = document.getElementById('inventoryTableBody');
      tbody.innerHTML = '';

      let lowStockCount = 0;
      let nearExpiryCount = 0;
      const today = new Date();

      inventory.forEach(item => {
        const expiryDate = new Date(item.expiry);
        const diffDays = Math.ceil((expiryDate - today) / (1000 * 60 * 60 * 24));

        const isLow = Number(item.quantity) <= Number(item.minThreshold);
        const isNearExpiry = diffDays <= 30 && diffDays >= 0;
        const isExpired = diffDays < 0;

        if (isLow) lowStockCount++;
        if (isNearExpiry || isExpired) nearExpiryCount++;

        const tr = document.createElement('tr');
        tr.className = 'hover:bg-slate-50 transition border-b border-slate-100';

        let badgeExpiry = '';
        if (isExpired) {
          badgeExpiry = `<span class="bg-rose-100 text-rose-700 px-2 py-0.5 rounded text-xs font-semibold">Telah Luput</span>`;
        } else if (isNearExpiry) {
          badgeExpiry = `<span class="bg-amber-100 text-amber-800 px-2 py-0.5 rounded text-xs font-semibold">${diffDays} hari lagi</span>`;
        }

        tr.innerHTML = `
          <td class="py-3 px-4 font-medium text-slate-900">${item.name}</td>
          <td class="py-3 px-4"><span class="bg-slate-100 text-slate-700 text-xs px-2 py-1 rounded font-medium">${item.category}</span></td>
          <td class="py-3 px-4 font-mono text-xs">${item.lot}</td>
          <td class="py-3 px-4">
            <div class="flex items-center gap-2">
              <span class="font-bold text-sm ${isLow ? 'text-rose-600' : 'text-slate-700'}">${item.quantity}</span>
              ${isLow ? '<span class="text-xs bg-rose-50 text-rose-600 border border-rose-200 px-1.5 py-0.5 rounded">Rendah</span>' : ''}
            </div>
          </td>
          <td class="py-3 px-4">
            <div class="text-xs">${item.expiry}</div>
            <div class="mt-0.5">${badgeExpiry}</div>
          </td>
          <td class="py-3 px-4 text-xs text-slate-500">
            <div><i class="fa-solid fa-snowflake text-cyan-600 mr-1"></i>${item.storage}</div>
            <div>${item.location || '-'}</div>
          </td>
          <td class="py-3 px-4 text-center">
            <div class="flex justify-center items-center gap-1">
              <button onclick="adjustQuantity('${item.id}', 1)" title="Guna 1 Pek (-)" class="w-7 h-7 bg-slate-100 hover:bg-slate-200 text-slate-700 rounded flex items-center justify-center text-xs transition">
                <i class="fa-solid fa-minus"></i>
              </button>
              <button onclick="adjustQuantity('${item.id}', -1)" title="Tambah 1 Pek (+)" class="w-7 h-7 bg-slate-100 hover:bg-slate-200 text-slate-700 rounded flex items-center justify-center text-xs transition">
                <i class="fa-solid fa-plus"></i>
              </button>
              <button onclick="deleteItem('${item.id}')" title="Padam Item" class="w-7 h-7 bg-rose-50 hover:bg-rose-100 text-rose-600 rounded flex items-center justify-center text-xs transition ml-2">
                <i class="fa-solid fa-trash"></i>
              </button>
            </div>
          </td>
        `;
        tbody.appendChild(tr);
      });

      // Update KPI
      document.getElementById('stat-total-items').textContent = inventory.length;
      document.getElementById('stat-low-stock').textContent = lowStockCount;
      document.getElementById('stat-near-expiry').textContent = nearExpiryCount;

      filterTable();
    }

    function adjustQuantity(id, changeDirection) {
      const item = inventory.find(i => i.id === id);
      if (!item) return;

      if (changeDirection === 1) { // Deduct 1 unit
        if (item.quantity > 0) {
          item.quantity = Number(item.quantity) - 1;
        } else {
          alert('Baki stok sudah kosong (0)!');
          return;
        }
      } else { // Add 1 unit
        item.quantity = Number(item.quantity) + 1;
      }
      saveInventory();
    }

    function deleteItem(id) {
      if (confirm('Adakah anda pasti mahu memadam rekod item ini?')) {
        inventory = inventory.filter(i => i.id !== id);
        saveInventory();
      }
    }

    function openModal() {
      document.getElementById('stockForm').reset();
      document.getElementById('itemId').value = '';
      document.getElementById('stockModal').classList.remove('hidden');
    }

    function closeModal() {
      document.getElementById('stockModal').classList.add('hidden');
    }

    function handleFormSubmit(e) {
      e.preventDefault();
      const newItem = {
        id: Date.now().toString(),
        name: document.getElementById('itemName').value,
        category: document.getElementById('itemCategory').value,
        lot: document.getElementById('itemLot').value,
        quantity: parseInt(document.getElementById('itemQuantity').value, 10),
        minThreshold: parseInt(document.getElementById('itemMinThreshold').value, 10),
        expiry: document.getElementById('itemExpiry').value,
        storage: document.getElementById('itemStorage').value,
        location: document.getElementById('itemLocation').value
      };

      inventory.push(newItem);
      saveInventory();
      closeModal();
    }

    function filterTable() {
      const search = document.getElementById('searchInput').value.toLowerCase();
      const category = document.getElementById('categoryFilter').value;
      const rows = document.querySelectorAll('#inventoryTableBody tr');

      let visibleCount = 0;
      rows.forEach(row => {
        const text = row.innerText.toLowerCase();
        const matchesSearch = text.includes(search);
        const matchesCategory = category === '' || text.includes(category.toLowerCase());

        if (matchesSearch && matchesCategory) {
          row.style.display = '';
          visibleCount++;
        } else {
          row.style.display = 'none';
        }
      });

      document.getElementById('emptyMessage').style.display = (visibleCount === 0) ? 'block' : 'none';
    }

    function exportToCSV() {
      if (inventory.length === 0) {
        alert('Tiada data untuk dieksport.');
        return;
      }
      let csvContent = "data:text/csv;charset=utf-8,";
      csvContent += "ID,Nama Item,Kategori,No Lot,Kuantiti Semasa,Paras Minima,Tarikh Luput,Suhu,Lokasi\\n";

      inventory.forEach(i => {
        csvContent += `"${i.id}","${i.name}","${i.category}","${i.lot}","${i.quantity}","${i.minThreshold}","${i.expiry}","${i.storage}","${i.location || ''}"\\n`;
      });

      const encodedUri = encodeURI(csvContent);
      const link = document.createElement("a");
      link.setAttribute("href", encodedUri);
      link.setAttribute("download", `Stok_Immunoassay_${new Date().toISOString().slice(0,10)}.csv`);
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
    }

    // Papar data permulaan semasa halaman dimuatkan
    window.onload = renderTable;
  </script>
